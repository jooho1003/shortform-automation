```python
def uses_chain(cuts: list[Cut]) -> bool:
    """끝 프레임 열이 하나라도 채워져 있으면 체인 기획안이다. 구형 기획안은 검증을 건너뛴다."""
    return any(c.end_action for c in cuts)


def _validate_chain(cuts: list[Cut], sets: dict[str, str]) -> None:
    """설계 10 §5. 체인이 깨지면 영상 경계가 안 이어지므로 파싱 단계에서 막는다."""
    if not uses_chain(cuts):
        return
    prev: Cut | None = None
    for c in cuts:
        if c.start_mode not in START_MODES:
            raise StepFailed(f"컷 {c.n} 시작 열은 {'/'.join(START_MODES)} 중 하나여야 합니다: {c.start_mode!r}")
        if not c.end_action:
            raise StepFailed(f"컷 {c.n} 끝 프레임이 비어 있습니다. 체인 기획안은 모든 컷에 끝 프레임이 필요합니다.")
        if not c.view:
            raise StepFailed(f"컷 {c.n} 뷰가 비어 있습니다.")
        if sets and c.view not in sets:
            raise StepFailed(f"컷 {c.n} 뷰 {c.view}가 '## 세트' 절에 없습니다.")
        if abs((c.end - c.start) - round(c.end - c.start)) > 1e-6:
            raise StepFailed(f"컷 {c.n} 길이가 정수 초가 아닙니다 ({c.end - c.start:g}초). 영상 모델이 정수 초만 받습니다.")
        if prev is None:
            if c.start_mode != "새로":
                raise StepFailed(f"첫 컷 {c.n}은 '새로'여야 합니다.")
        else:
            if abs(c.start - prev.end) > 1e-6:
                raise StepFailed(f"컷 {prev.n}→{c.n} TC가 이어지지 않습니다 ({prev.end:g} → {c.start:g}).")
            if c.chained and c.view != prev.view:
                raise StepFailed(f"컷 {c.n} '이어서'는 앞 컷과 같은 뷰여야 합니다 ({prev.view} → {c.view}).")
        prev = c
```

```python
    - '이어서' 컷의 시작은 앞 컷의 끝 파일이므로 (n,start)는 (n-1,end)로 옮긴다.
    - 어떤 컷의 끝은 다음 '이어서' 컷의 시작이므로, 그 컷의 끝도 바뀐다. 체인 끝까지 번진다.
    결과는 (컷, start|end) 정렬 목록. start가 end보다 앞."""
    by_n = {c.n: c for c in plan.cuts}
    todo = list(frames)
    out: set[tuple[int, str]] = set()
    while todo:
        n, which = todo.pop()
        cut = by_n[n]
        if which == "start" and cut.chained:
            todo.append((n - 1, "end"))
            continue
        if (n, which) in out:
            continue
        out.add((n, which))
        if which == "start":
            todo.append((n, "end"))
        else:
            nxt = by_n.get(n + 1)
            if nxt is not None and nxt.chained:
                todo.append((n + 1, "end"))
    return sorted(out, key=lambda f: (f[0], 0 if f[1] == "start" else 1))
```

`_parse_fix`를 통째로 아래로 교체(반환형이 프레임 목록으로 바뀐다):
```python
def _parse_fix(note: str, known_cuts: set[int]) -> tuple[bool, list[tuple[int, str]], str]:
    """Return (replace_sheet, frames, instruction). frames = [(컷, 'start'|'end')] 정렬.

    v2: {"version":2,"frames":[{"cut":3,"which":"start"}],"instruction":"…"} — 프레임 단위(설계 10 §5).
    v1: {"version":1,"cuts":[3],"instruction":"…"}, 번호 목록 '1, 3', '시트' — 컷의 시작·끝 둘 다."""
    stripped = note.strip()
    if not stripped:
        raise _fix_error("요청이 비어 있습니다.")

    def both(cuts: list[int]) -> list[tuple[int, str]]:
        return sorted((n, w) for n in cuts for w in ("start", "end"))

    if stripped == "시트":
        return True, both(sorted(known_cuts)), "시트"
    if stripped.startswith("{"):
        try:
            value = json.loads(stripped)
        except json.JSONDecodeError as exc:
            raise _fix_error("JSON을 읽을 수 없습니다.") from exc
        if not isinstance(value, dict):
            raise _fix_error("JSON 객체여야 합니다.")
        instruction = value.get("instruction")
        if not isinstance(instruction, str) or not instruction.strip():
            raise _fix_error("instruction은 비어 있지 않은 문자열이어야 합니다.")
        if value.get("version") == 2 and set(value) == {"version", "frames", "instruction"}:
            frames = value.get("frames")
            if not isinstance(frames, list) or not frames:
                raise _fix_error("frames는 비어 있지 않은 배열이어야 합니다.")
            parsed: set[tuple[int, str]] = set()
            for item in frames:
                if not isinstance(item, dict) or type(item.get("cut")) is not int or item.get("which") not in ("start", "end"):
                    raise _fix_error('frames 항목은 {"cut": 번호, "which": "start"|"end"} 여야 합니다.')
                if item["cut"] not in known_cuts:
                    raise _fix_error("frames에는 현재 컷표에 있는 번호만 넣으세요.")
                parsed.add((item["cut"], item["which"]))
            return False, sorted(parsed), instruction.strip()
        if set(value) != {"version", "cuts", "instruction"}:
            raise _fix_error("version, cuts(또는 v2의 frames), instruction 세 필드만 필요합니다.")
        cuts = value.get("cuts")
        if type(value.get("version")) is not int or value.get("version") != 1 or not isinstance(cuts, list):
            raise _fix_error("version은 1, cuts는 배열이어야 합니다.")
        if any(type(n) is not int or n <= 0 for n in cuts) or len(set(cuts)) != len(cuts):
            raise _fix_error("cuts에는 중복 없는 양의 정수만 넣으세요.")
        if not cuts or any(n not in known_cuts for n in cuts):
            raise _fix_error("cuts에는 현재 컷표에 있는 번호만 넣으세요.")
        return False, both(sorted(cuts)), instruction.strip()
    if re.fullmatch(r"\d+(?:\s*,\s*\d+)*", stripped):
        cuts = [int(n.strip()) for n in stripped.split(",")]
        if len(set(cuts)) != len(cuts) or any(n not in known_cuts for n in cuts):
            raise _fix_error("번호 목록에 중복 또는 알 수 없는 컷이 있습니다.")
        return False, both(cuts), ""
    first, *rest = stripped.splitlines()
    m = re.fullmatch(r"\s*(\d+)\s*번\s*컷\s*[:：\-–—]?\s*(.+?)\s*", first)
    if m and not rest:
        n = int(m.group(1))
        if n in known_cuts:
            return False, both([n]), m.group(2)
    raise _fix_error("대상을 판별할 수 없습니다.")
```

```python
class FrameJob:

    out: Path

def _frame_jobs(ep: Episode, plan, targets, sheet: Path, extra: list[Path], hint: str, instruction_for) -> list[FrameJob]:
    """만들어야 할 프레임을 컷 순서로 늘어놓는다. 이미 있는 파일은 건너뛴다.

    for cut in sorted(targets, key=lambda c: c.n):
        instr = instruction_for(cut.n)
        start = ep.clips_dir / KEYFRAME.format(n=cut.n)
        if not cut.chained and not start.exists():
            refs = [sheet, *extra]
            pose_ref = None
            prev = by_n.get(cut.n - 1)
            if cut.start_mode == "각도변경" and prev is not None:

                jobs.append(FrameJob(cut, "end", end, [sheet, *extra, start_ref], _end_frame_prompt(plan, cut, start_ref, end, hint, instr, extra)))
    return jobs
```

```python
def _estimate(cfg: dict, cuts) -> float:
    """초당 단가가 있으면 길이 합으로, 없으면 컷당 고정가로."""
    per_sec = cfg.get("cost_per_second_usd")
    if per_sec is not None:
        return sum(c.seconds for c in cuts) * float(per_sec)
    return len(cuts) * float(cfg.get("cost_per_cut_usd", 0.6))


def _check_duration(cfg: dict, cut) -> None:
    lo, hi = cfg.get("duration_min_sec"), cfg.get("duration_max_sec")
    if (lo is not None and cut.seconds < float(lo)) or (hi is not None and cut.seconds > float(hi)):
        raise StepFailed(f"컷 {cut.n} 길이 {cut.seconds:g}초가 모델 허용 범위({lo}~{hi}초) 밖입니다. 컷표를 고치세요.")


def _needs_end_frame(cfg: dict) -> bool:
    return "{end_keyframe}" in json.dumps(cfg.get("body", {}), ensure_ascii=False)


def _check_coverage(windows: list[tuple[float, float]], total: float, max_gap: float = 0.6) -> None:
    """설계 10 §1: 모든 프레임에 대사가 있어야 한다. 겹침과 긴 공백은 돈 쓰기 전에 멈춘다."""
    prev_end = 0.0
    for i, (start, end) in enumerate(windows, 1):
        if start < prev_end - 0.05:
            raise StepFailed(f"{i}번 대사가 앞 대사와 겹칩니다 ({start:.2f} < {prev_end:.2f}). 대사 시각을 뒤로 미루세요.")
        if start - prev_end > max_gap:
            raise StepFailed(f"{i}번 대사 앞에 {start - prev_end:.2f}초 공백이 있습니다 (허용 {max_gap}초). 대사 시각을 당기세요.")
        prev_end = max(prev_end, end)
    if total - prev_end > max_gap + 0.05:
        raise StepFailed(f"마지막 대사가 끝난 뒤 {total - prev_end:.2f}초가 비어 있습니다 (허용 {max_gap}초). 끝 컷을 줄이거나 대사를 미루세요.")
```

```python
def _dialogue_payload(plan, windows: list[tuple[float, float]] | None = None) -> list[dict]:
    items = [
        {"at": d.at, "character": d.character, "text": d.text, "direction": getattr(d, "direction", ""),
         "english": getattr(d, "english", "")}
        for d in plan.dialogues
    ]
    # 실제 발화 구간. 웹 스토리보드의 대사 띠가 읽는다. 렌더는 at만 쓴다.

    return items
```

```python
def _synthesize(ep: Episode, plan, voice: dict) -> list[tuple[Path, float, float]]:
    """대사마다 mp3 하나를 만들고 (경로, 발화 시작, 발화 끝)을 편 기준 절대초로 돌려준다."""
    tts_dir = ep.assets_dir / "tts"
    tts_dir.mkdir(parents=True, exist_ok=True)
    out: list[tuple[Path, float, float]] = []
    for index, dialogue in enumerate(plan.dialogues, 1):
        path = tts_dir / f"dialogue-{index:02d}.mp3"
        begin, finish = asyncio.run(_speak(dialogue.text, path, _voice_for(voice, getattr(dialogue, "character", ""))))
        if finish <= begin:
            # 문장 경계 이벤트가 없으면 파일 길이로 대신한다. 0초 구간이면 공백 검사가 엉뚱하게 실패한다.
            begin, finish = 0.0, audio_duration(path)
        out.append((path, dialogue.at + begin, dialogue.at + finish))
    return out
```

```python
def run(ep: Episode) -> None:
    plan = planfile_a.load(ep.plan_md)
    cfg = _config()
    raw_dir = ep.assets_dir / "원본"
    clip_dir = ep.assets_dir / "클립"
    raw_dir.mkdir(parents=True, exist_ok=True)
    clip_dir.mkdir(parents=True, exist_ok=True)
    fail_log = ep.assets_dir / "실패.txt"
    sheet = ep.clips_dir / SHEET

    # 1) 대사 먼저(무료). 발화 구간을 재서 모든 프레임에 대사가 있는지 확인한 뒤에 돈을 쓴다.
    synthesized: list[tuple[Path, float, float]] = []
    if plan.dialogues:
        print("  대사 합성 중...")
        synthesized = _synthesize(ep, plan, load_voice(ep, freeze=True))
        _check_coverage([(s, e) for _, s, e in synthesized], plan.total_seconds)

    # 2) 비용·길이 가드
    todo = [c for c in plan.cuts if not (raw_dir / f"{c.n:02d}.mp4").exists() and not (clip_dir / f"{c.n:02d}.mp4").exists()]
    for cut in todo:
        _check_duration(cfg, cut)
    est = _estimate(cfg, todo)
    cap = float(cfg.get("max_cost_per_episode_usd", 10))
    spent = float(state.load(ep).get("fal_spent_usd", 0.0))
    print(f"  생성할 컷 {len(todo)}개 · 예상 ${est:.2f} (누적 ${spent:.2f}, 상한 ${cap:.0f})")
    if todo:
        if spent + est > cap:
            raise StepFailed(f"편당 비용 상한 ${cap:.0f}를 넘깁니다 (누적 ${spent:.2f} + 예상 ${est:.2f}). config/fal-chain.yaml 을 조정하거나 컷을 줄이세요.")
        key = os.environ.get("FAL_KEY")
        if not key:
            raise MissingKey("FAL_KEY 가 없습니다. https://fal.ai/dashboard/keys 에서 발급해 pipeline/.env 에 넣으세요.")

    # 3) 컷 영상
    failed = []
    for cut in todo:
        start = ep.clips_dir / start_frame_name(plan, cut)
        end = ep.clips_dir / end_frame_name(cut)
        if not start.exists():
            print(f"  ! 컷 {cut.n:02d} 시작 프레임 없음({start.name}) — 건너뜀")
            failed.append(cut.n)
            continue
        if _needs_end_frame(cfg) and not end.exists():
            print(f"  ! 컷 {cut.n:02d} 끝 프레임 없음({end.name}) — 건너뜀. 설정이 끝 프레임을 요구합니다.")
            failed.append(cut.n)
            continue
        prompt = _video_prompt(plan, cut)
        print(f"  컷 {cut.n:02d} 생성 (fal, {cut.seconds:g}초) — {prompt[:70]}")
        try:
            fal.generate(cfg, key, raw_dir / f"{cut.n:02d}.mp4", {
                "prompt": prompt,
                "keyframe": fal.data_uri(start),
                "end_keyframe": fal.data_uri(end) if end.exists() else "",
                "seconds": f"{cut.seconds:g}",
                "sheet": fal.data_uri(sheet) if sheet.exists() else "",
            })
            spent += _estimate(cfg, [cut])
            st = state.load(ep)
            st["fal_spent_usd"] = round(spent, 2)
            state.save(ep, st)
        except StepFailed as exc:
            print(f"  ! 컷 {cut.n:02d} 실패: {exc}")
            failed.append(cut.n)
```

````markdown
## 조사 참고

- 1화는 시리즈의 출발점이다. 레퍼런스 구조(상대역이 말을 걸고 주인공이 답한 뒤 속마음)는 7~11컷에서 쓰고, 1~6컷은 상상 장면으로 캐릭터의 환상을 보여 준다.
- 상상과 현실은 조명으로 가른다. 상상은 황금빛과 부드러운 광륜, 현실은 락 블록의 사무실 조명이다.
- “뭐든 시켜 주십시오”가 이후 모든 잡일의 원인이다. 그래서 이 편만 마지막 대사가 “이것도 경험이다”가 아니다.
````

```ts
/**
 * 스토리보드 순수 헬퍼. 파일 이름·체인 규칙은 pipeline/shortform/steps/check_frames.py 와 같아야 한다.
 * 화면은 회차 폴더의 파일 목록과 기획안 JSON만 보고 그린다.
 */
import type {PlanADialogue, PlanCut} from './types';

export type SpeechWindow = {index: number; character: string; text: string; start: number; end: number};
export type FrameRef = {cut: number; which: 'start' | 'end'};
type TimelineDialogue = {at: number; speechStart?: number; speechEnd?: number};

const pad = (n: number) => String(n).padStart(2, '0');

export const endFrameName = (cut: PlanCut) => `키프레임-${pad(cut.n)}-끝.png`;

/** '이어서' 컷은 자기 시작 파일이 없다. 앞 컷의 끝 프레임이 시작이다. */
export function startFrameName(cuts: PlanCut[], cut: PlanCut): string {
  if (cut.start_mode !== '이어서') return `키프레임-${pad(cut.n)}.png`;
  const prev = cuts.filter((c) => c.n < cut.n).sort((a, b) => b.n - a.n)[0];
  return prev ? endFrameName(prev) : `키프레임-${pad(cut.n)}.png`;
}

export function chainGroups(cuts: PlanCut[]): PlanCut[][] {
  const groups: PlanCut[][] = [];
  for (const cut of [...cuts].sort((a, b) => a.n - b.n)) {
    if (cut.start_mode === '이어서' && groups.length) groups[groups.length - 1].push(cut);
    else groups.push([cut]);
  }
  return groups;
}

const SYLLABLES_PER_SECOND = 5; // 설계 09 실측: 레퍼런스 3.5~5.2음절/초, 사구 TTS +8%
const EDGE_LEAD_SECONDS = 0.2;  // Edge mp3 앞 무음

export function speechWindows(dialogues: PlanADialogue[], timeline: TimelineDialogue[] | null | undefined): {windows: SpeechWindow[]; estimated: boolean} {
  const measured = timeline?.filter((d) => typeof d.speechStart === 'number' && typeof d.speechEnd === 'number') ?? [];
  const estimated = measured.length !== dialogues.length;
  const windows = dialogues.map((d, index) => {
    const hit = estimated ? undefined : measured.find((m) => Math.abs(m.at - d.at) < 0.001);
    if (hit) return {index, character: d.character, text: d.text, start: hit.speechStart as number, end: hit.speechEnd as number};
    const syllables = d.text.replace(/[\s…,.!?~'"“”‘’]/g, '').length;
    const start = Math.round((d.at + EDGE_LEAD_SECONDS) * 10) / 10;
    return {index, character: d.character, text: d.text, start, end: Math.round((start + syllables / SYLLABLES_PER_SECOND) * 10) / 10};
  });
  return {windows, estimated};
}

/** 대사 없는 구간. 설계 10 §1: 0.6초를 넘으면 안 된다. 앞·사이·뒤를 모두 본다. */
export function coverageGaps(windows: {start: number; end: number}[], total: number, maxGap = 0.6): {from: number; to: number}[] {
  const gaps: {from: number; to: number}[] = [];
  let prevEnd = 0;
  for (const w of [...windows].sort((a, b) => a.start - b.start)) {
    if (w.start - prevEnd > maxGap) gaps.push({from: prevEnd, to: w.start});
    prevEnd = Math.max(prevEnd, w.end);
  }
  if (total - prevEnd > maxGap) gaps.push({from: prevEnd, to: total});
  return gaps;
}

const frameOrder = (a: FrameRef, b: FrameRef) => a.cut - b.cut || (a.which === 'start' ? -1 : 1);

export function fixRequestNote(frames: FrameRef[], instruction: string): string {
  const unique = Array.from(new Map(frames.map((f) => [`${f.cut}-${f.which}`, f])).values()).sort(frameOrder);
  return JSON.stringify({version: 2, frames: unique, instruction: instruction.trim()});
}

export function formatFixRequest(note: string): string {
  try {
    const value = JSON.parse(note) as {cuts?: unknown; frames?: unknown; instruction?: unknown};
    if (Array.isArray(value.frames) && typeof value.instruction === 'string') {
      const parts = (value.frames as FrameRef[]).map((f) => `컷 ${f.cut} ${f.which === 'start' ? '시작' : '끝'}`);
      return `${parts.join(', ')}: ${value.instruction}`;
    }
    if (Array.isArray(value.cuts) && typeof value.instruction === 'string') return `컷 ${value.cuts.join(', ')}: ${value.instruction}`;
  } catch {
    // 기존 일반 텍스트 요청도 그대로 보여준다.
  }
  return note;
}

/**
 * 프레임을 다시 만들면 영상이 바뀌는 컷. 파이썬 expand_frames 와 같은 규칙:
 * 시작이 바뀌면 그 컷의 끝도 바뀌고, 끝은 다음 '이어서' 컷의 시작이므로 그 컷의 끝도 바뀐다.
 */
export function affectedCuts(cuts: PlanCut[], frames: FrameRef[]): number[] {
  const byN = new Map(cuts.map((c) => [c.n, c]));
  const changed = new Set<string>();
  const todo = [...frames];
  while (todo.length) {
    const f = todo.pop() as FrameRef;
    const cut = byN.get(f.cut);
    if (!cut) continue;
    if (f.which === 'start' && cut.start_mode === '이어서') { todo.push({cut: f.cut - 1, which: 'end'}); continue; }
    const key = `${f.cut}-${f.which}`;
    if (changed.has(key)) continue;
    changed.add(key);
    if (f.which === 'start') todo.push({cut: f.cut, which: 'end'});
    else if (byN.get(f.cut + 1)?.start_mode === '이어서') todo.push({cut: f.cut + 1, which: 'end'});
  }
  return Array.from(new Set(Array.from(changed).map((k) => Number(k.split('-')[0])))).sort((a, b) => a - b);
}
```
